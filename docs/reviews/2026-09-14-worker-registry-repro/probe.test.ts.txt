// Review probes assert the observed baseline, including unsafe behavior.
// They are evidence for a design review, not regression tests for a fix.
import { afterAll, beforeAll, expect, it } from 'vitest'
import { randomUUID } from 'node:crypto'
import { openContractDb } from '../../packages/db/src/__tests__/contract-fixture.js'
import {
  registerWorker,
  getWorkerById,
  heartbeatWorker,
  markLostWorkers,
  markWorkerDraining,
  markWorkerStopped,
} from '../../packages/db/src/leases/leases.js'
import { assertSchemaReady } from '../../packages/db/src/migrate-native.js'
import { assertWorkerEndpointAllowed, workerStatusSchema } from '@cairn/shared'

let handle: Awaited<ReturnType<typeof openContractDb>>
beforeAll(async () => { handle = await openContractDb('sqlite') })
afterAll(async () => { await handle?.close() })

async function seed(workerId: string, lostAfterSeconds = 120) {
  const input = { workerId, instanceId: randomUUID(), capacity: 1, lostAfterSeconds }
  await registerWorker(handle.db, input)
  return input
}

async function age(workerId: string, seconds: number) {
  await handle.raw('UPDATE workers SET heartbeat_at = ? WHERE id = ?', [
    new Date(Date.now() - seconds * 1000).toISOString(), workerId,
  ])
}

it('a candidate with a shorter timeout can take over before the incumbent timeout', async () => {
  const incumbent = await seed('timeout-takeover', 120)
  await age(incumbent.workerId, 90)
  await expect(registerWorker(handle.db, { ...incumbent, instanceId: randomUUID() }))
    .rejects.toMatchObject({ code: 'WORKER_ID_CONFLICT' })
  const candidate = { ...incumbent, instanceId: randomUUID(), lostAfterSeconds: 60 }
  await registerWorker(handle.db, candidate)
  expect((await getWorkerById(handle.db, incumbent.workerId))?.instanceId).toBe(candidate.instanceId)
})

it('a reaper uses its own timeout, not the timeout used by the registered worker', async () => {
  const incumbent = await seed('timeout-reaper', 120)
  await age(incumbent.workerId, 90)
  expect(await markLostWorkers(handle.db, 60)).toContain(incumbent.workerId)
  expect((await getWorkerById(handle.db, incumbent.workerId))?.status).toBe('LOST')
})

it('stale shutdown writes can mark the replacement instance DRAINING and STOPPED', async () => {
  const incumbent = await seed('shutdown-fence', 60)
  await age(incumbent.workerId, 180)
  const replacement = { ...incumbent, instanceId: randomUUID() }
  await registerWorker(handle.db, replacement)
  expect(await heartbeatWorker(handle.db, incumbent.workerId, incumbent.instanceId)).toBe('instance_taken')
  // These are the exact unfenced calls made by onApplicationShutdown.
  await markWorkerDraining(handle.db, incumbent.workerId)
  expect(await getWorkerById(handle.db, incumbent.workerId))
    .toMatchObject({ instanceId: replacement.instanceId, status: 'DRAINING' })
  await markWorkerStopped(handle.db, incumbent.workerId)
  expect(await getWorkerById(handle.db, incumbent.workerId))
    .toMatchObject({ instanceId: replacement.instanceId, status: 'STOPPED' })
  expect(await heartbeatWorker(handle.db, replacement.workerId, replacement.instanceId)).toBe('lost')
  expect(workerStatusSchema.safeParse('STOPPED').success).toBe(true)
})

it('the current endpoint validator accepts non-origins and wildcard HTTPS hosts', () => {
  for (const value of [
    'https://0.0.0.0:8091',
    'https://[::]:8091',
    'https://worker.invalid:8091/prefix?mode=test#fragment',
    'https://example-user:example-password@worker.invalid:8091',
    'http://127.0.0.1:9222',
  ]) {
    expect(() => assertWorkerEndpointAllowed(value, 'production')).not.toThrow()
  }
  expect(() => assertWorkerEndpointAllowed('http://worker.invalid:8091', 'production')).toThrow()
})

it('startup validation rejects a database with an additional migration', async () => {
  await assertSchemaReady(handle, handle.env)
  await handle.raw('INSERT INTO _migrations VALUES (?, ?, ?, ?)',
    ['9999', '9999_review_future.sql', 'review-placeholder', 'complete'])
  try {
    await expect(assertSchemaReady(handle, handle.env)).rejects.toThrow('incompatible')
  } finally {
    await handle.raw('DELETE FROM _migrations WHERE prefix = ?', ['9999'])
  }
})
